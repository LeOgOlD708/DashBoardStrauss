+function computeRotation() {
+  const prices = state.data.prices || {};

+  const out = [];

+    const n = Math.min(sec.hist.length, spy.hist.length);
+    if (n < WEEK * (SMA_W + MOM_W + TAIL)) continue; // ~180 barras minimo

+    for (let i = n - 1; i >= 0; i -= WEEK) {
+      if (pH[i] > 0) ratio.unshift(sH[i] / pH[i]);
+    }
+    if (ratio.length < SMA_W + MOM_W + 2) continue;
+    // RS-Ratio: ratio vs su SMA de 26 muestras semanales
+    const rsr = [];
+    for (let i = SMA_W - 1; i < ratio.length; i++) {
+      let sum = 0;
+      for (let j = i - SMA_W + 1; j <= i; j++) sum += ratio[j];
+      const sma = sum / SMA_W;
+      rsr.push(sma > 0 ? 100 * ratio[i] / sma : 100);
+    }
+    // RS-Momentum: RSR vs RSR de hace 4 semanas
+    const pts = [];
+    for (let i = MOM_W; i < rsr.length; i++) {
+      if (rsr[i - MOM_W] > 0) pts.push({ x: rsr[i], y: 100 * rsr[i] / rsr[i - MOM_W] });
+    }
+    if (pts.length < 2) continue;
+    const tail = pts.slice(-TAIL);

+               : last.x <  100 && last.y >= 100 ? 'mejorando' : 'rezagado';
+    out.push({ tk, name, tail, last, quad });
+  }
+  return out.length ? out : null;
